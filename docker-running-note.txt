INSERT INTO items (
    item_category_id,
    tax_id,
    name,
    slug,
    caution,
    description,
    price,
    price_with_tax,
    status,
    item_type,
    tax_name,
    tax_rate,
    tax_type,
    tax_amount,
    `order`,
    is_featured,
    creator_type,
    creator_id,
    editor_type,
    editor_id,
    kitchen_printer_id,
    label_printer_id,
    barcode,
    manage_stock,
    is_print_menu,
    is_print_label,
    branch_id,
    created_at,
    updated_at
)
SELECT
    1 AS item_category_id,   -- set new category
    tax_id,
    name,
    CONCAT(slug, '-copy-', UNIX_TIMESTAMP()) AS slug, -- prevent duplicate slug
    caution,
    description,
    price,
    price_with_tax,
    status,
    item_type,
    tax_name,
    tax_rate,
    tax_type,
    tax_amount,
    `order`,
    is_featured,
    creator_type,
    creator_id,
    editor_type,
    editor_id,
    kitchen_printer_id,
    label_printer_id,
    barcode,
    manage_stock,
    is_print_menu,
    is_print_label,
    1 AS branch_id,          -- set new branch
    NOW(),
    NOW()
FROM items
WHERE name LIKE 'S%'
  AND branch_id = 2;
